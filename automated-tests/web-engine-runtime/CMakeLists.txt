cmake_minimum_required(VERSION 3.16)
project(web-engine-runtime-tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(DALI_API_MAJOR 3 CACHE STRING "Installed DALi API major")
find_package(PkgConfig REQUIRED)
pkg_check_modules(DALI REQUIRED IMPORTED_TARGET dali${DALI_API_MAJOR}-adaptor)
enable_testing()
foreach(backend chromium lwe)
  add_library(${backend} SHARED fixture.cpp)
  target_link_libraries(${backend} PRIVATE PkgConfig::DALI)
  set_target_properties(${backend} PROPERTIES OUTPUT_NAME dali${DALI_API_MAJOR}-web-engine-${backend}-plugin)
endforeach()
add_executable(runtime-tests runtime-tests.cpp)
target_link_libraries(runtime-tests PRIVATE PkgConfig::DALI ${CMAKE_DL_LIBS})
target_compile_options(runtime-tests PRIVATE -UNDEBUG)
target_compile_definitions(runtime-tests PRIVATE TEST_PLUGIN_NAME="libdali${DALI_API_MAJOR}-web-engine-chromium-plugin.so")
foreach(scenario transfer failure lifetime unsupported)
  add_test(NAME runtime-${scenario} COMMAND runtime-tests ${scenario})
  set_tests_properties(runtime-${scenario} PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=${CMAKE_CURRENT_BINARY_DIR}:$ENV{LD_LIBRARY_PATH}")
endforeach()

target_compile_definitions(runtime-tests PRIVATE TEST_LWE_PLUGIN_NAME="libdali${DALI_API_MAJOR}-web-engine-lwe-plugin.so")
# Separate processes and search paths model both installed default symlinks.
foreach(backend chromium lwe)
  if(backend STREQUAL "chromium")
    set(backend_type 0)
  else()
    set(backend_type 1)
  endif()
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${backend}-default")
  file(CREATE_LINK "../libdali${DALI_API_MAJOR}-web-engine-${backend}-plugin.so"
    "${CMAKE_CURRENT_BINARY_DIR}/${backend}-default/libdali${DALI_API_MAJOR}-web-engine-plugin.so" SYMBOLIC)
  foreach(selection environment symlink)
    if(selection STREQUAL "environment")
      set(environment_name ${backend})
    else()
      set(environment_name symlink)
    endif()
    foreach(direction default-first explicit-first)
      set(scenario alias-${backend}-${selection}-${direction})
      add_test(NAME runtime-${scenario} COMMAND runtime-tests ${scenario} ${backend_type} ${direction} ${environment_name})
      set_tests_properties(runtime-${scenario} PROPERTIES ENVIRONMENT
        "LD_LIBRARY_PATH=${CMAKE_CURRENT_BINARY_DIR}/${backend}-default:${CMAKE_CURRENT_BINARY_DIR}:$ENV{LD_LIBRARY_PATH}")
    endforeach()
  endforeach()
endforeach()
