IF( (NOT UBUNTU_PROFILE) )
  SET(APPLICATION_WIDGET_TARGET ${name}-application-widget )
  SET(APPLICATION_COMPONENT_BASED_TARGET ${name}-application-component-based )

  MESSAGE(STATUS ${ROOT_SRC_DIR}/dali/internal/application-model )
  SET( appmodel_src_dir ${ROOT_SRC_DIR}/dali/internal/application-model )

  INCLUDE( ${appmodel_src_dir}/file.list )

  SET( APPLICATION_WIDGET_SOURCES ${app_widget_src_files} )
  SET( APPLICATION_COMPONENT_BASED_SOURCES ${app_component_based_src_files} )

  ADD_LIBRARY( ${APPLICATION_WIDGET_TARGET} SHARED ${APPLICATION_WIDGET_SOURCES} )
  ADD_LIBRARY( ${APPLICATION_COMPONENT_BASED_TARGET} SHARED ${APPLICATION_COMPONENT_BASED_SOURCES} )

  OPTION(ENABLE_PKG_CONFIGURE "Use pkgconfig" ON)

  IF (ENABLE_PKG_CONFIGURE)
    FIND_PACKAGE(PkgConfig REQUIRED)
    IF( USE_TCORE_BACKEND )
      PKG_CHECK_MODULES(APPCOREUICPP app-core-tcore-cpp)
    ELSE()
      PKG_CHECK_MODULES(APPCOREUICPP app-core-ui-cpp)
    ENDIF()
    PKG_CHECK_MODULES(APPCORECOMMONCPP appcore-common)
    PKG_CHECK_MODULES(ELDBUS eldbus)

    SET(CORE_PKG_CFG_FILE ${name}-application-model.pc)
    CONFIGURE_FILE(${CMAKE_CURRENT_LIST_DIR}/dali-adaptor-application-model.pc.in ${CORE_PKG_CFG_FILE} @ONLY)
  ENDIF()

  IF( USE_TCORE_BACKEND )
    SET( APPLICATION_MODEL_EXTRA_LDFLAGS ${TIZEN_CORE_LDFLAGS} ${TIZEN_CORE_WL_LDFLAGS} )
    SET( APPLICATION_MODEL_EXTRA_CFLAGS ${TIZEN_CORE_CFLAGS} ${TIZEN_CORE_WL_CFLAGS} )
  ELSE()
    SET( APPLICATION_MODEL_EXTRA_LDFLAGS "" )
    SET( APPLICATION_MODEL_EXTRA_CFLAGS "" )
  ENDIF()

  TARGET_LINK_LIBRARIES( ${APPLICATION_WIDGET_TARGET} PUBLIC
    ${DLOG_LDFLAGS}
    ${DALICORE_LDFLAGS}
    ${ELDBUS_LDFLAGS}
    ${CAPI_APPFW_WIDGET_BASE_LDFLAGS}
    ${APPLICATION_MODEL_EXTRA_LDFLAGS}
    ${name}
  )

  TARGET_COMPILE_OPTIONS( ${APPLICATION_WIDGET_TARGET} PUBLIC
    ${DLOG_CFLAGS}
    ${DALICORE_CFLAGS}
    ${ELDBUS_CFLAGS}
    ${CAPI_APPFW_WIDGET_BASE_CFLAGS}
    ${APPLICATION_MODEL_EXTRA_CFLAGS}
    -I../../../
    -Wall
    # widget_base_* APIs are deprecated on newer Tizen but must still be called
    # to keep widget application support; they must not fail -Werror builds.
    -Wno-deprecated-declarations
  )

  TARGET_LINK_LIBRARIES( ${APPLICATION_COMPONENT_BASED_TARGET} PUBLIC
    ${DLOG_LDFLAGS}
    ${DALICORE_LDFLAGS}
    ${ELDBUS_LDFLAGS}
    ${COMPONENT_BASED_CORE_BASE_LDFLAGS}
    ${APPLICATION_MODEL_EXTRA_LDFLAGS}
    ${name}
  )

  TARGET_COMPILE_OPTIONS( ${APPLICATION_COMPONENT_BASED_TARGET} PUBLIC
    ${DLOG_CFLAGS}
    ${DALICORE_CFLAGS}
    ${ELDBUS_CFLAGS}
    ${COMPONENT_BASED_CORE_BASE_CFLAGS}
    ${APPLICATION_MODEL_EXTRA_CFLAGS}
    -I../../../
    -Wall
  )

  INSTALL( TARGETS ${APPLICATION_WIDGET_TARGET} DESTINATION ${LIB_DIR} )
  INSTALL( TARGETS ${APPLICATION_COMPONENT_BASED_TARGET} DESTINATION ${LIB_DIR} )
ENDIF()
